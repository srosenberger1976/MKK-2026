function openSignaturePopup(side) { const btn = document.getElementById("btn_sig_" + side); if (btn && btn.classList.contains("row-disabled")) { alert("Hinweis: Du kannst erst unterschreiben, wenn alle Spiele des Spielplans erfasst wurden!"); return; } const overlay = document.getElementById("popup_sig_" + side); overlay.style.display = "flex"; document.body.style.overflow = "hidden"; // Der 40ms-Timeout sorgt dafür, dass das Feld auf Anhieb geladen und beschreibbar ist setTimeout(() => { const canvas = document.getElementById("canvas_sig_" + side); sigCanvases[side] = canvas; const ctx = canvas.getContext("2d"); sigContexts[side] = ctx; let exactWidth = canvas.parentElement.clientWidth; let exactHeight = canvas.parentElement.clientHeight; if (!exactWidth || exactWidth <= 0) exactWidth = 320; if (!exactHeight || exactHeight <= 0) exactHeight = 200; canvas.style.width = exactWidth + "px"; canvas.style.height = exactHeight + "px"; const pixelRatio = 3; canvas.width = exactWidth * pixelRatio; canvas.height = exactHeight * pixelRatio; ctx.scale(pixelRatio, pixelRatio); ctx.strokeStyle = "#000000"; ctx.lineWidth = 3.5; ctx.lineCap = "round"; ctx.lineJoin = "round"; setupCanvasEvents(side); }, 40); }
